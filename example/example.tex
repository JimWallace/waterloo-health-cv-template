% Example CV for the template, with fictional content. It exercises every
% part of the template (each section style, the biblatex categories, the
% \hl... macros, grant and supervision records, an APR-only note) so the template can be
% developed and checked without anyone's real CV.
%
% Compile example.tex for the plain CV, example-apr.tex for the APR version.
\input{uwhealthcv-pdfa}% PDF/A output (template/); must come before \documentclass
\documentclass[11pt]{article}

\usepackage{uwhealthcv}

\addbibresource{example.bib}
\setreviewperiod{2025}{2026}
\cvowner{Alex Example}

\DeclareBibliographyCategory{journal}
\addtocategory{journal}{example2026journal, example2019journal}
\DeclareBibliographyCategory{conference}
\addtocategory{conference}{example2025conference, example2026meetup}
\DeclareBibliographyCategory{technicalreports}
\addtocategory{technicalreports}{example2021report}
\DeclareBibliographyCategory{software}
\addtocategory{software}{example2026software}
% A line of detail under one entry (e.g. generated development stats)
\setentrydetail{example2026software}{Developed January -- June 2026: 12 releases, 340 commits and 18,000 lines of code.}
% Counts a script writes from the record (usually \input from a generated file)
\setcvnumber{review-works}{3}
\setcvnumber{review-works-trainee-first}{2}
\DeclareBibliographyCategory{presentations}
\addtocategory{presentations}{example2026talk}
\DeclareBibliographyCategory{invited}
\addtocategory{invited}{example2025invited}

% APR colours: trainee work in two colours; software as teaching first;
% conference leadership as service and research.
\addtocategory{apralsoteaching}{example2026journal, example2026talk, example2026posteraward}
\addtocategory{aprteaching}{example2026software}
\addtocategory{apralsoresearch}{example2026software, exampleconf2026chair}
% chairing an examination: teaching and service
\addtocategory{apralsoservice}{candidate2026chair}
% a meetup co-organized at a conference: service first, with teaching
% (mentorship) and research; two apralso... lists give a third colour,
% which the dots style shows
\addtocategory{aprservice}{example2026meetup}
\addtocategory{apralsoteaching}{example2026meetup}
\addtocategory{apralsoresearch}{example2026meetup}
