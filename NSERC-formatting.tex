% The page of the Faculty of Health's vitae format (the .doc linked from
% uwaterloo.ca/health/faculty-staff/curriculum-vitae), whose settings are in
% twips (1/1440in): left 1440, right 1526, top 1195 (to the body text),
% bottom 778, header 720 from the top of the page, footer 245 from the
% bottom. The header line (11pt) is 13.6pt high, so the gap below it makes
% up the rest of the top margin. The type is 11pt Times (TeX Gyre Termes
% below): a CV uses \documentclass[11pt]{article}.
\usepackage[letterpaper,
  left=1in, right=1.059722in,     % 1440, 1526 twips
  top=0.829861in, bottom=0.540278in, % 1195, 778 twips
  headheight=13.6pt,
  headsep=\dimexpr0.329861in-13.6pt\relax, % header at 720 twips
  footskip=0.370139in]{geometry}  % footer at 245 twips
\usepackage{mathptmx}% Times for maths (and for text under pdfLaTeX)
% LuaLaTeX (the template's engine) has no Times text font from mathptmx;
% TeX Gyre Termes is the same design as an OpenType font.
\usepackage{iftex}
\ifLuaTeX\usepackage{fontspec}\setmainfont{TeX Gyre Termes}\fi
\usepackage{wasysym}
\usepackage{pifont}
\usepackage{url}
\usepackage{hyperref}

    colorlinks=true,        % Enable colored links

    urlcolor=black,          % Color of URLs
    citecolor=black,        % Color of citations
    linkbordercolor=white,  % Set the border color to white (effectively hides it)
    pdfborder={0 0 0}       % Remove the border
}
